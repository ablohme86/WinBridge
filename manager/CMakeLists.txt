cmake_minimum_required(VERSION 3.16)
project(WinBridgeManager VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTORCC ON)
add_executable(winbridge-manager main.cpp i18n.qrc)
target_link_libraries(winbridge-manager PRIVATE Qt6::Widgets)
install(TARGETS winbridge-manager RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    set(CMAKE_AUTOMOC ON)
    add_executable(manager-tests test_manager.cpp i18n.qrc)
    target_link_libraries(manager-tests PRIVATE Qt6::Widgets Qt6::Test)
    add_test(NAME manager-ui COMMAND manager-tests -platform offscreen)
endif()
